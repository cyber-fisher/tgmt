**2. 核心代码移植（手动修改）**

从 `idea_graph/support_txt/TreeTxtActiveGraphSimplifier.cpp` 抽取以下函数，插入到新文件：

| 函数名 | 起始行号（参考） | 功能 |
|--------|----------------|------|
| `buildActiveSubgraph` | ~350-450 | 构建活性边集 |
| `computeJaccardDistance` | ~470 | Jaccard 距离 |
| `computeGnnDistance` | ~540-700 | GNN 向量余弦距离（含确定性权重生成） |
| `applyGcnLayer` | ~650 | GCN 层前向传播 |
| `deterministicWeight` | ~580 | He 初始化权重生成 |

**3. 替换度量接口**

在 `TreeTxtMultiBranchActiveGraphSimplifier.cpp` 中找到：
```cpp
// 原代码（v2）：基于 VTI 体积 IoU
double computeBranchLocalVolumeDistance(...) {
    auto vol1 = buildBranchLocalVolumeAtScalar(...);
    auto vol2 = buildBranchLocalVolumeAtScalar(...);
    return 1.0 - computeIoU(vol1, vol2);
}
```

替换为（新代码）：
```cpp
// 新代码：基于活性子图 Jaccard/GNN
double computeBranchLocalActiveGraphDistance(
    int nodeA, int nodeB, int branchIdx,
    const std::string& similarityType,  // "jaccard" or "gnn"
    int gnnHiddenDim = 64
) {
    auto subgraphA = buildActiveSubgraph(nodeA, branchIdx);
    auto subgraphB = buildActiveSubgraph(nodeB, branchIdx);
    
    if (similarityType == "jaccard") {
        return computeJaccardDistance(subgraphA.edges, subgraphB.edges);
    } else if (similarityType == "gnn") {
        return computeGnnDistance(subgraphA, subgraphB, gnnHiddenDim);
    }
    return 1.0;
}
```

```cpp
for (int i = 1; i < argc; ++i) {
```

```python
import os
import glob

def check_structural_nodes(stats_file):
    """从 stats.txt 中读取结构节点保留情况"""
    with open(stats_file) as f:
        lines = f.readlines()
    
    # 假设格式：original_nodes=420, kept_nodes=XX, ...
    # 以及 critical_nodes=YY（你需要在代码中输出这一行）
    original = kept = critical = 0
    for line in lines:
        if 'original_nodes=' in line:
            original = int(line.split('=')[1])
        elif 'kept_nodes=' in line:
            kept = int(line.split('=')[1])
        elif 'critical_nodes=' in line:
            critical = int(line.split('=')[1])
    
    # 验证：kept >= critical（结构节点应全部保留）
    return kept >= critical, original, kept, critical

results_dir = "experiments/results/exp1_topology"
all_passed = True

for stats_file in glob.glob(f"{results_dir}/*_stats.txt"):
    passed, orig, kept, crit = check_structural_nodes(stats_file)
    status = "✓" if passed else "✗"
    print(f"{status} {os.path.basename(stats_file)}: {orig} nodes, {kept} kept, {crit} critical")
    if not passed:
        all_passed = False

print("\n" + ("="*50))
if all_passed:
    print("✓ Exp 1 验证通过：所有结构节点均保留")
else:
    print("✗ Exp 1 验证失败：存在结构节点未保留的情况")
```

```python
import glob
import re

def parse_stats(stats_file):
    with open(stats_file) as f:
        text = f.read()
    eps = float(re.search(r'eps=([\d.]+)', text).group(1))
    orig = int(re.search(r'original_nodes=(\d+)', text).group(1))
    kept = int(re.search(r'kept_nodes=(\d+)', text).group(1))
    compression = (orig - kept) / orig * 100
    return eps, compression

results_dir = "experiments/results/exp2_epsilon_sweep"

# case17
case17_data = []
for f in sorted(glob.glob(f"{results_dir}/case17_eps*_stats.txt")):
    eps, comp = parse_stats(f)
    case17_data.append((eps, comp))
    print(f"case17 eps={eps:.2f} -> compression={comp:.1f}%")

# 保存为 CSV
import csv
with open(f"{results_dir}/case17_epsilon_curve.csv", 'w', newline='') as csvfile:
    writer = csv.writer(csvfile)
    writer.writerow(['epsilon', 'compression_rate'])
    writer.writerows(case17_data)

print(f"\n✓ 数据已保存到 {results_dir}/case17_epsilon_curve.csv")
```

### 3.3 计算 GeomErr（需要在代码中新增输出）

**代码修改（在合并后的简化器中）：**

在 `simplifyWithMultiBranchWeightedDistance` 函数里，对每个**被跳过**的候选节点 `cand`，记录它与当前参考节点 `ref` 的距离 `d`：

```cpp
// 伪代码
if (d < epsilon) {
    // 跳过该节点
    skippedDistances.push_back(d);  // 新增：记录被跳过节点的距离
} else {
    // 保留该节点
    keptNodes.push_back(cand);
    ref = cand;
}
```

在 `writeStats` 函数中输出：
```cpp
// 在 stats.txt 末尾追加
double meanGeomErr = std::accumulate(skippedDistances.begin(), skippedDistances.end(), 0.0) / skippedDistances.size();
std::sort(skippedDistances.begin(), skippedDistances.end());
double p90GeomErr = skippedDistances[static_cast<size_t>(skippedDistances.size() * 0.9)];

statsFile << "geom_err_mean=" << meanGeomErr << "\n";
statsFile << "geom_err_p90=" << p90GeomErr << "\n";
```

```python
import re

def extract_geom_err(stats_file):
    with open(stats_file) as f:
        text = f.read()
    mean = float(re.search(r'geom_err_mean=([\d.]+)', text).group(1))
    p90 = float(re.search(r'geom_err_p90=([\d.]+)', text).group(1))

# ISM

print(f"| Jaccard | {jac_mean:.4f} | {jac_p90:.4f} |")
print(f"| GNN | {gnn_mean:.4f} | {gnn_p90:.4f} |")
```
